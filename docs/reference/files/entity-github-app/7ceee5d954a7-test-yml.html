name: test

  push:
  pull_request:

jobs:
  test:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.10'
      - name: Locate OpenSSL
        shell: pwsh
        run: |
          $openssl = (Get-Command openssl).Source
          "ENTITY_GITHUB_OPENSSL=$openssl" | Out-File -FilePath $env:GITHUB_ENV -Encoding utf8 -Append
      - name: Run tests
        shell: pwsh
        run: |
          $env:PYTHONPATH = "$PWD\src"
          python -m unittest discover -s tests -v
