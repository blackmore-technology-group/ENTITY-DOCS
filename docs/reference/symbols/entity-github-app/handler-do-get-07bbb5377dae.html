    def do_GET(self) -> None:
        if self.path != "/health":
            self._json(404, {"ok": False, "error": "not_found"})
            return
        store = EvidenceStore(self.server.db_path)
        try:
            self._json(200, {"ok": True, "service": "entity-github-app", "store": store.counts(), "audit": store.verify_audit()})
        finally:
            store.close()
