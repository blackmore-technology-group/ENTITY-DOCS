    def _storage_path(self,stored: str,kind: str) -> Path:
        raw=Path(str(stored))
        if not raw.is_absolute(): return self.root/raw
        if raw.exists(): return raw
        base=self.keys if kind=="key" else self.objects
        return base/raw.name
